def allocation():
 seen=set();logical=allocated=files=0
 for area in (ROOT,BOOT,STATE,TEMP,WORKSPACE/'a',WORKSPACE/'b'):
  for base,dirs,names in os.walk(area,followlinks=False):
   for name in names:
    try:s=(pathlib.Path(base)/name).lstat()
    except FileNotFoundError:continue
    if not stat.S_ISREG(s.st_mode):continue
    key=(s.st_dev,s.st_ino)
    if key in seen:continue
    seen.add(key);logical+=s.st_size;allocated+=s.st_blocks*512;files+=1
 return {'logical_bytes':logical,'allocated_bytes':allocated,'files':files}

libproc=ctypes.CDLL('/usr/lib/libproc.dylib',use_errno=True)
libproc.proc_pid_rusage.argtypes=[ctypes.c_int,ctypes.c_int,ctypes.c_void_p]
libproc.proc_pid_rusage.restype=ctypes.c_int
def physical(pid):
 buffer=ctypes.create_string_buffer(96)
 if libproc.proc_pid_rusage(pid,0,buffer)!=0:return 0
 # Darwin rusage_info_v0: UUID[16], then seven uint64_t fields.
 return int.from_bytes(buffer.raw[72:80],sys.byteorder)
assert physical(os.getpid())>0

def rss_tree(roots):
 out=subprocess.check_output(['ps','-axo','pid=,ppid=,rss='],text=True)
 rows=[list(map(int,line.split())) for line in out.splitlines() if line.strip()]
 owned=set(roots)
 while True:
  old=len(owned);owned.update(pid for pid,parent,rss in rows if parent in owned)
  if len(owned)==old:break
 return sum(rss for pid,parent,rss in rows if pid in owned)*1024,sum(physical(pid) for pid in owned)

results=[]

def run(name,command,env,cwd,limit=1200,extra_pids=()):
 start=time.monotonic();reason=None;next_sample=0;peak_allocated=peak_rss=peak_physical=0;next_memory=0;rss=footprint=0
 with (ROOT/'logs'/f'{name}.log').open('w') as log:
  proc=subprocess.Popen(command,env=env,cwd=cwd,stdout=log,stderr=log,start_new_session=True)
  while proc.poll() is None:
   elapsed=time.monotonic()-start;s=os.statvfs(ROOT);free=s.f_bavail*s.f_frsize
   if elapsed>=limit:reason=f'{limit} second phase limit'
   if mode.startswith('blaze') and time.monotonic()-workflow_start>=2700:reason='45 minute candidate workflow limit'
   if free<48*1024**3:reason='48 GiB available-space floor'
   if elapsed>=next_memory:
    rss,footprint=rss_tree((proc.pid,*extra_pids));peak_rss=max(peak_rss,rss);peak_physical=max(peak_physical,footprint);next_memory=elapsed+2
    if footprint>20*1024**3:reason='20 GiB sampled aggregate physical-footprint limit'
   if elapsed>=next_sample:
    a=allocation();peak_allocated=max(peak_allocated,a['allocated_bytes']);next_sample=elapsed+20
    with (ROOT/'logs/monitor.jsonl').open('a') as f:f.write(json.dumps({'phase':name,'wall_seconds':elapsed,'available_bytes':free,'owned_rss_bytes':rss,'owned_physical_bytes':footprint,**a})+'\n')
    if a['allocated_bytes']>64*1024**3:reason='64 GiB summed allocated-byte allowance'
    if rss>20*1024**3:reason='20 GiB sampled aggregate RSS limit'
   if reason:
    try:os.killpg(proc.pid,signal.SIGTERM)
    except ProcessLookupError:pass
    try:proc.wait(timeout=15)
    except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);proc.wait()
    break
   time.sleep(.25)
 row={'name':name,'command':command,'cwd':str(cwd),'exit':proc.returncode,'wall_seconds':round(time.monotonic()-start,3),'stopped_by':reason,'peak_sampled_allocated_bytes':peak_allocated,'peak_sampled_rss_bytes':peak_rss,'peak_sampled_physical_bytes':peak_physical,'final_inventory':allocation()}
 results.append(row);(ROOT/'logs'/f'{mode}-results.json').write_text(json.dumps(results,indent=2)+'\n');print({k:row[k] for k in ('name','exit','wall_seconds','stopped_by')},flush=True)
 if reason:raise RuntimeError(reason)
 return row
