def allocation():
 seen=set();logical=allocated=files=0
 for area in (ROOT,WORKSPACE/'a',WORKSPACE/'b',WORKSPACE/'c'):
  for base,dirs,names in os.walk(area,followlinks=False):
   for name in names:
    try:s=(pathlib.Path(base)/name).lstat()
    except FileNotFoundError:continue
    if not stat.S_ISREG(s.st_mode):continue
    key=(s.st_dev,s.st_ino)
    if key in seen:continue
    seen.add(key);logical+=s.st_size;allocated+=s.st_blocks*512;files+=1
 return {'logical_bytes':logical,'allocated_bytes':allocated,'files':files}

libproc=ctypes.CDLL('/usr/lib/libproc.dylib',use_errno=True)

def physical(pid):
 buffer=ctypes.create_string_buffer(96)
 if libproc.proc_pid_rusage(pid,0,buffer)!=0:return 0
 # Darwin rusage_info_v0: UUID[16], then seven uint64_t fields.

def rss_tree(roots):
 out=subprocess.check_output(['ps','-axo','pid=,ppid=,rss='],text=True)
 rows=[list(map(int,line.split())) for line in out.splitlines() if line.strip()]
 owned=set(roots)
 while True:
  old=len(owned);owned.update(pid for pid,parent,rss in rows if parent in owned)
  if len(owned)==old:break
 return sum(rss for pid,parent,rss in rows if pid in owned)*1024,sum(physical(pid) for pid in owned)
